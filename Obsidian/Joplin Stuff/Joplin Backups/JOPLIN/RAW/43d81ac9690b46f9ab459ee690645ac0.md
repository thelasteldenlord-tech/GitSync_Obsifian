        document.querySelectorAll('.nav-folder-title').forEach(title => {

                // Close all other folders
                document.querySelectorAll('.nav-folder').forEach(f => {
                    f.classList.remove('open');
                });
                
                // Toggle current folder
                if (!wasOpen) {
                    folder.classList.add('open');
                }
            });
        });
